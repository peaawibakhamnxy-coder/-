function toHex(bytes) {

function generateSecMsGecV2() {
  let ticks = Math.floor(Date.now() / 1000) + WIN_EPOCH;
  ticks -= ticks % 300;
  const ticksStr = (BigInt(ticks) * 10000000n).toString();
  const strToHash = new TextEncoder().encode(ticksStr + TRUSTED_CLIENT_TOKEN);
  const digest = sha256Digest(strToHash);
  const pair = cs64Digest(strToHash, digest);
  const b = new Uint8Array(8);
  const bdv = new DataView(b.buffer);
  bdv.setUint32(0, pair[0], true);
  bdv.setUint32(4, pair[1], true);
  return toHex(b) + '-' + toHex(digest);
}
